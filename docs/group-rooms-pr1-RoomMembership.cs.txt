namespace MusicSync.Core;

internal static class RoomMembership
{
    public static ConnectionView Update(ConnectionView view, ServerMessage message)
    {
        var room = message.Room ?? view.Room
            ?? throw new InvalidDataException("Expected a room snapshot.");
        var selfId = message.ParticipantId ?? view.SelfParticipantId;
        var participants = room.Participants.Select(p =>
            new PublicParticipant(p.ParticipantId, p.Role, p.Connected, p.Report)).ToArray();
        var self = participants.FirstOrDefault(p => p.ParticipantId == selfId);
        if (selfId is null || selfId == Guid.Empty || self is null)
            throw new InvalidDataException("Expected own public participant identity.");
        return view with
        {
            Participants = participants,
            SelfParticipantId = selfId,
            Capacity = room.Capacity,
            Role = self.Role
        };
    }

    public static PlayerReport? LeaderReport(ConnectionView view) =>
        view.Participants.FirstOrDefault(p => p.Role == Role.Leader)?.Report;

    public static bool? PeerConnected(ConnectionView view) =>
        view.Participants.Any(p => p.ParticipantId != view.SelfParticipantId && p.Connected);
}
